#
# restoHack (Hack 1.0.3) as a bare-metal RP2040 program, on rp2040-console.
#
# This is a separate build from restoHack's own top-level CMakeLists: that one
# detects curses and builds a Unix binary, neither of which applies on a Pico.
# Here there is no curses and no OS - the console supplies the terminal, the
# filesystem and libc's syscall surface, so all that is left is the game's own
# sources, which go to rp2040_console_game() unchanged.
#
# It is kept beside restoHack rather than inside it because the sources are
# reached as ../src; nothing is copied.
#
#   cmake -S . -B build \
#         -DGAME_CONSOLE=/path/to/rp2040-console \
#         -DPICO_SDK_PATH=/path/to/pico_sdk \
#         -DPICO_BOARD=pico -DCMAKE_BUILD_TYPE=Release
#   cmake --build build --target hack        # -> build/hack.uf2
#   cmake --build build --target volume      # -> build/hack-volume.uf2
#
# A 2MB Pico is what the flash layout below assumes: the volume sits at
# 0x180000 (1.5MB) and is 512KB, ending exactly at the 2MB mark.
#
cmake_minimum_required(VERSION 3.13)

# Where rp2040-console lives. Its bundled pico_sdk_import.cmake (before
# project()) and its cmake module (after) are both found here; PICO_SDK_PATH
# is read from the environment or the cache by that import.
if(NOT GAME_CONSOLE AND DEFINED ENV{GAME_CONSOLE})
    set(GAME_CONSOLE "$ENV{GAME_CONSOLE}")
endif()
if(NOT GAME_CONSOLE)
    message(FATAL_ERROR
        "set GAME_CONSOLE to the rp2040-console checkout, e.g.\n"
        "  cmake -DGAME_CONSOLE=../../rp2040-console ...")
endif()
get_filename_component(GAME_CONSOLE "${GAME_CONSOLE}" ABSOLUTE)

include(${GAME_CONSOLE}/pico_sdk_import.cmake)   # installs the ARM toolchain

project(hack C CXX ASM)
set(CMAKE_C_STANDARD 11)

include(${GAME_CONSOLE}/cmake/rp2040-console.cmake)   # calls pico_sdk_init()

set(HACK_ROOT  "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(HACK_SRC   "${HACK_ROOT}/src")
set(GEN        "${CMAKE_CURRENT_BINARY_DIR}/generated")

# --- config.h ---------------------------------------------------------
#
# hack.h includes <config.h>, which restoHack generates from config.h.in.
# On a desktop that file records what configure found; none of it applies
# here except getentropy(), which the console really does provide (from the
# RP2040 ring oscillator), so it alone is on and everything else is off.
#
set(HAVE_GETENTROPY ON)
configure_file("${HACK_ROOT}/config.h.in" "${GEN}/config.h")

# --- hack.onames.h ----------------------------------------------------
#
# Generated by makedefs. Cross-built, makedefs cannot be run as a target
# binary, so gen-onames.cmake builds it with the host compiler instead; see
# that file.
#
if(NOT HOST_C_COMPILER)
    find_program(HOST_C_COMPILER cc)
endif()

set(ONAMES "${GEN}/hack.onames.h")
add_custom_command(
    OUTPUT  ${ONAMES}
    COMMAND ${CMAKE_COMMAND}
            -DHOST_C_COMPILER=${HOST_C_COMPILER}
            -DMAKEDEFS_SRC=${HACK_SRC}/makedefs.c
            -DDEF_OBJECTS=${HACK_SRC}/def.objects.h
            -DMAKEDEFS_BIN=${CMAKE_CURRENT_BINARY_DIR}/makedefs-host
            -DOUT_FILE=${ONAMES}
            -P ${CMAKE_CURRENT_SOURCE_DIR}/gen-onames.cmake
    DEPENDS ${HACK_SRC}/makedefs.c ${HACK_SRC}/def.objects.h
            ${CMAKE_CURRENT_SOURCE_DIR}/gen-onames.cmake
    COMMENT "Generating hack.onames.h with makedefs"
)

# --- the game's sources -----------------------------------------------
#
# Everything in restoHack/src except makedefs - a host tool, run separately
# to make hack.onames.h, and never part of the game. This is the same set
# restoHack builds for Unix, termcap front end included: the console speaks
# termcap, so there is no curses path to compile in.
#
file(GLOB HACK_SOURCES "${HACK_SRC}/*.c")
list(REMOVE_ITEM HACK_SOURCES "${HACK_SRC}/makedefs.c")

rp2040_console_game(
    NAME        hack
    ROOT        ${HACK_ROOT}
    SOURCES     ${HACK_SOURCES}
    GENERATED   ${ONAMES}
    INCLUDE_DIRS ${HACK_SRC} ${GEN}
    LEGACY_C

    # The volume is the whole filesystem as far as Hack is concerned, mounted
    # at "/", so HackDIR is the root rather than the /usr/games/hack of a Unix
    # install. Its data files are read relative to it.
    ENV         TERM=xterm HACKDIR=/

    # Read-only game data baked into the volume.
    DATA_FILES  data help hh rumors

    # Files Hack expects to find and open, but does not create itself: the
    # score file and the two lock files. Absent, a game cannot finish.
    EMPTY_FILES record perm hack

    # Where save files and bones go.
    DATA_DIRS   save

    VOLUME_NAME HACKFS

    # Hack has a one-argument itoa() and its own panic(), both of which collide
    # with something in the link; the prefix header renames them out of the way.
    RENAMES     itoa=hackitoa panic=hack_panic
)
